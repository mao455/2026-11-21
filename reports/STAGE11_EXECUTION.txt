新检出的共享源码可生成可移植复跑计划（这是当前源码复跑，不能宣称历史 controller 字节相同）：

python - <<'PLAN'
import json
from pathlib import Path
here=Path('experiments/rax_dexbin_tinyobj')
for source,dest in [('STAGE11_SCALE_FINAL_PLAN.json','/tmp/stage11_scale_rerun.json'),('STAGE11_PRESSURE_FINAL_PLAN.json','/tmp/stage11_pressure_rerun.json')]:
    plan=json.loads((here/source).read_text())
    plan['rerun_source']='current shared source, fast/two, one attempt; historical declaration unchanged'
    plan.pop('controller_sha256',None)
    for row in plan['runs']:
        row.pop('controller',None)
        row.update(profile='fast',fingers='two',max_attempts=1)
    Path(dest).write_text(json.dumps(plan,indent=2))
PLAN
